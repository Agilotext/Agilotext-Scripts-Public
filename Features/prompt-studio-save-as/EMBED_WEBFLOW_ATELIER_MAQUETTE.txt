<script>
  window.__AGILO_PROMPT_STUDIO__ = {
    enabled: true,
    readOnly: false,
    editHtml: true,
    studioMode: "expert",
    showPreviewTab: true,
    showFieldList: true,
    showConsistencyTab: true,
    apiBase: "https://api.agilotext.com/api/v1",
    mountSelector: "#agilo-prompt-studio-anchor"
  };

  document.addEventListener("DOMContentLoaded", function () {
    document.addEventListener("click", function (e) {
      var btn = e.target.closest(".view-edit-btn");
      if (!btn) return;
      var row = btn.closest("[data-prompt-id]");
      if (!row) return;
      var id = row.getAttribute("data-prompt-id");
      if (!id || !window.AgiloPromptStudio || !window.AgiloPromptStudio.openModalAndSelect) return;
      e.preventDefault();
      e.stopPropagation();
      window.AgiloPromptStudio.openModalAndSelect(String(id));
    }, true);

    if (window.AgiloPromptStudio) window.AgiloPromptStudio.init();
  });
</script>
